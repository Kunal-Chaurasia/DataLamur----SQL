select s1.session_id, count(s2.session_id) as concurrent_sessions
from session as s1
inner join sessions as s2
	 on s1.session_id != s2.session_id
	 and ((s1.start_time between s2.start_time and s2.end_time) or (s2.start_time between s1.start_time and s1.end_time))
group by s1.session_id
order by concurrent_sessions desc
