with orders as (
select to_char(order_placed_time, 'YYYY-MM') as year_month, delivery_id,
	case when extract(minute from (actual_delivery_time - predicted_delivery_time)) > 20 then 1 else 0 end as extremely_late
from delivery_orders
)

select year_month, sum(extremely_late)/count(distinct delivery_id)::float * 100 as perc_extremely_late
from orders
group by 1;
